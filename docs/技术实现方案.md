### 1.4 AI服务选型

#### 1.4.1 多模态大语言模型（核心）

采用具有视觉识别能力的多模态大语言模型，统一处理图像识别和智能对话：

| 服务商 | 模型名称 | 能力 | 推荐指数 |
|-------|---------|-----|---------|
| 百度智能云 | 文心一言 (ERNIE-VL) | 图像理解、对话生成、营养分析 | ⭐⭐⭐⭐⭐ |
| 腾讯云 | 混元大模型 (多模态版) | 图像识别、智能对话、内容生成 | ⭐⭐⭐⭐⭐ |
| 阿里云 | 通义千问-VL | 视觉理解、多轮对话、知识问答 | ⭐⭐⭐⭐ |
| 智谱AI | GLM-4V | 图像理解、对话生成 | ⭐⭐⭐⭐ |

#### 1.4.2 选型优势

使用多模态大语言模型的优势：

| 优势 | 说明 |
|-----|-----|
| 统一接口 | 图像识别和对话生成使用同一API，简化开发 |
| 上下文理解 | 模型可理解对话历史，提供连贯的多轮交互 |
| 灵活输出 | 可自定义输出格式，便于结构化数据提取 |
| 智能分析 | 不仅识别食物，还能分析烹饪方式、估算份量 |
| 个性化建议 | 结合用户档案和饮食历史，生成个性化建议 |
| 成本优化 | 减少API调用次数，降低整体成本 |

```javascript
exports.main = async (event, context) => {
  const { action, data, provider = 'baidu' } = event
  
  try {
    let result
    
    switch (action) {
      case 'foodRecognition':
        result = await handleFoodRecognition(data, provider)
        break
      case 'nutritionAnalysis':
        result = await handleNutritionAnalysis(data, provider)
        break
      case 'recommendation':
        result = await handleRecommendation(data, provider)
        break
      case 'chat':
        result = await handleChat(data, provider)
        break
      default:
        throw new Error(`Unknown action: ${action}`)
    }
    
    return {
      success: true,
      data: result
    }
  } catch (error) {
    console.error('AI Gateway Error:', error)
    
    if (provider === 'baidu') {
      return await handleFallback(event, error)
    }
    
    return {
      success: false,
      error: error.message
    }
  }
}

async function handleFoodRecognition(data, provider) {
  const { imageUrl } = data
  const prompt = PROMPT_TEMPLATES.foodRecognition
  
  const response = await callMultimodalAI(provider, {
    prompt: prompt,
    image: imageUrl
  })
  
  return parseJSONResponse(response)
}

async function handleNutritionAnalysis(data, provider) {
  const { foodInfo, userInfo } = data
  const prompt = PROMPT_TEMPLATES.nutritionAnalysis
    .replace('{foodInfo}', JSON.stringify(foodInfo, null, 2))
    .replace('{userInfo}', JSON.stringify(userInfo, null, 2))
  
  const response = await callTextAI(provider, { prompt })
  return parseJSONResponse(response)
}

async function handleRecommendation(data, provider) {
  const { userInfo, todayRecords, nutrientGaps } = data
  const prompt = PROMPT_TEMPLATES.recommendation
    .replace('{userInfo}', JSON.stringify(userInfo, null, 2))
    .replace('{todayRecords}', JSON.stringify(todayRecords, null, 2))
    .replace('{nutrientGaps}', JSON.stringify(nutrientGaps, null, 2))
  
  const response = await callTextAI(provider, { prompt })
  return parseJSONResponse(response)
}

async function handleChat(data, provider) {
  const { message, conversationHistory, userProfile } = data
  const prompt = PROMPT_TEMPLATES.chat
    .replace('{conversationHistory}', formatConversationHistory(conversationHistory))
    .replace('{userProfile}', JSON.stringify(userProfile, null, 2))
  
  const response = await callTextAI(provider, { 
    prompt: prompt,
    userMessage: message 
  })
  
  return { message: response }
}

async function callMultimodalAI(provider, params) {
  const config = AI_CONFIG[provider]
  
  switch (provider) {
    case 'baidu':
      return await callBaiduMultimodal(config, params)
    case 'tencent':
      return await callTencentMultimodal(config, params)
    case 'aliyun':
      return await callAliyunMultimodal(config, params)
    default:
      throw new Error(`Unsupported provider: ${provider}`)
  }
}

async function callBaiduMultimodal(config, params) {
  const accessToken = await getBaiduAccessToken(config.apiKey, config.secretKey)
  
  const response = await axios.post(
    `${config.endpoint}?access_token=${accessToken}`,
    {
      messages: [
        {
          role: 'user',
          content: [
            { type: 'text', text: params.prompt },
            { type: 'image', image: params.image }
          ]
        }
      ]
    }
  )
  
  return response.data.result
}

async function callTencentMultimodal(config, params) {
  const tencentcloud = require('tencentcloud-sdk-nodejs')
  const HunyuanClient = tencentcloud.hunyuan.v20230901.Client
  
  const client = new HunyuanClient({
    credential: {
      secretId: config.secretId,
      secretKey: config.secretKey
    },
    region: 'ap-guangzhou'
  })
  
  const response = await client.ChatCompletions({
    Model: config.model,
    Messages: [
      {
        Role: 'user',
        Contents: [
          { Type: 'text', Text: params.prompt },
          { Type: 'image_url', ImageUrl: { Url: params.image } }
        ]
      }
    ]
  })
  
  return response.Choices[0].Message.Content
}

async function callAliyunMultimodal(config, params) {
  const response = await axios.post(
    config.endpoint,
    {
      model: config.model,
      input: {
        messages: [
          {
            role: 'user',
            content: [
              { image: params.image },
              { text: params.prompt }
            ]
          }
        ]
      }
    },
    {
      headers: {
        'Authorization': `Bearer ${config.apiKey}`,
        'Content-Type': 'application/json'
      }
    }
  )
  
  return response.data.output.choices[0].message.content
}

async function getBaiduAccessToken(apiKey, secretKey) {
  const response = await axios.get(
    `https://aip.baidubce.com/oauth/2.0/token?grant_type=client_credentials&client_id=${apiKey}&client_secret=${secretKey}`
  )
  return response.data.access_token
}

function parseJSONResponse(text) {
  try {
    const jsonMatch = text.match(/\{[\s\S]*\}/)
    if (jsonMatch) {
      return JSON.parse(jsonMatch[0])
    }
    throw new Error('No valid JSON found in response')
  } catch (error) {
    console.error('JSON Parse Error:', error)
    throw new Error('Failed to parse AI response')
  }
}

function formatConversationHistory(history) {
  if (!history || history.length === 0) {
    return '暂无历史对话'
  }
  
  return history.map(msg => 
    `${msg.role === 'user' ? '用户' : 'AI'}: ${msg.content}`
  ).join('\n')
}

async function handleFallback(event, originalError) {
  console.log('Attempting fallback to Tencent Hunyuan...')
  
  try {
    const fallbackEvent = { ...event, provider: 'tencent' }
    return await exports.main(fallbackEvent, context)
  } catch (fallbackError) {
    console.error('Fallback also failed:', fallbackError)
    return {
      success: false,
      error: 'All AI services are currently unavailable'
    }
  }
}
```

```javascript
export function calculateNutrition(foodName, weight) {
  const food = FOOD_DATABASE[foodName]
  if (!food) return null
  
  const ratio = weight / 100
  const nutrition = {}
  
  for (const [key, value] of Object.entries(food.nutrition)) {
    nutrition[key] = Math.round(value * ratio * 10) / 10
  }
  
  return {
    name: foodName,
    category: food.category,
    weight: weight,
    nutrition: nutrition
  }
}

export function calculateDailyTargets(userInfo) {
  const { gender, age, height, weight, activityLevel, goal } = userInfo
  
  const BMR = gender === 'male' 
    ? 10 * weight + 6.25 * height - 5 * age + 5
    : 10 * weight + 6.25 * height - 5 * age - 161
  
  const activityMultipliers = {
    sedentary: 1.2,
    light: 1.375,
    moderate: 1.55,
    active: 1.725
  }
  
  const TDEE = BMR * activityMultipliers[activityLevel]
  
  const goalAdjustments = {
    lose: -300,
    maintain: 0,
    gain: 300
  }
  
  const targetCalories = Math.round(TDEE + goalAdjustments[goal])
  
  return {
    calories: targetCalories,
    protein: Math.round(targetCalories * 0.15 / 4),
    fat: Math.round(targetCalories * 0.25 / 9),
    carbohydrate: Math.round(targetCalories * 0.55 / 4),
    fiber: 25,
    water: 2000
  }
}
```

#### 2.3.2 营养缺口分析

```javascript
// utils/nutritionAnalyzer.js

export function analyzeNutrientGaps(actual, targets) {
  const gaps = {}
  const surplus = {}
  const adequate = {}
  
  for (const nutrient in targets) {
    const actualValue = actual[nutrient] || 0
    const targetValue = targets[nutrient]
    const ratio = actualValue / targetValue
    
    const result = {
      actual: Math.round(actualValue * 10) / 10,
      target: targetValue,
      ratio: Math.round(ratio * 100),
      unit: getNutrientUnit(nutrient)
    }
    
    if (ratio < 0.8) {
      result.gap = Math.round((targetValue - actualValue) * 10) / 10
      result.severity = ratio < 0.5 ? 'high' : 'medium'
      result.status = 'deficient'
      gaps[nutrient] = result
    } else if (ratio > 1.1) {
      result.surplus = Math.round((actualValue - targetValue) * 10) / 10
      result.severity = ratio > 1.3 ? 'high' : 'medium'
      result.status = 'excess'
      surplus[nutrient] = result
    } else {
      result.status = 'adequate'
      adequate[nutrient] = result
    }
  }
  
  return { gaps, surplus, adequate }
}

function getNutrientUnit(nutrient) {
  const units = {
    calories: 'kcal',
    protein: 'g',
    fat: 'g',
    carbohydrate: 'g',
    fiber: 'g',
    vitaminC: 'mg',
    vitaminA: 'μg',
    calcium: 'mg',
    iron: 'mg',
    sodium: 'mg'
  }
  return units[nutrient] || 'g'
}

export function getNutrientName(nutrient) {
  const names = {
    calories: '热量',
    protein: '蛋白质',
    fat: '脂肪',
    carbohydrate: '碳水化合物',
    fiber: '膳食纤维',
    vitaminC: '维生素C',
    vitaminA: '维生素A',
    calcium: '钙',
    iron: '铁',
    sodium: '钠'
  }
  return names[nutrient] || nutrient
}
```

```javascript
        sourceType: ['camera', 'album'],
        sizeType: ['compressed']
      })

        }
      }

    }
```

### 7.2 日志记录

```javascript
// 云函数日志记录

const cloud = require('wx-server-sdk')

cloud.init()

exports.main = async (event, context) => {
  const startTime = Date.now()
  const wxContext = cloud.getWXContext()
  
  try {
    // 业务逻辑
    const result = await doSomething(event)
    
    // 记录成功日志
    console.log({
      level: 'INFO',
      action: event.action,
      userId: wxContext.OPENID,
      duration: Date.now() - startTime,
      success: true
    })
    
    return result
  } catch (error) {
    // 记录错误日志
    console.error({
      level: 'ERROR',
      action: event.action,
      userId: wxContext.OPENID,
      duration: Date.now() - startTime,
      error: error.message,
      stack: error.stack
    })
    
    throw error
  }
}
```
