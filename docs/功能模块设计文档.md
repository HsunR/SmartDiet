```javascript
export const INTENTS = {
  RECORD_FOOD: 'record_food',       // 记录食物

  ],
```

**本地意图识别**：

```javascript
// utils/intentRecognizer.js

import { INTENTS, INTENT_PATTERNS } from '../constants/intents'

export function recognizeIntent(message) {
  for (const [intent, patterns] of Object.entries(INTENT_PATTERNS)) {
    for (const pattern of patterns) {
      if (pattern.test(message)) {
        return {
          intent,
          confidence: 0.8,
          matched: true
        }
      }
    }
  }
  
  return {
    intent: INTENTS.UNKNOWN,
    confidence: 0,
    matched: false
  }
}

export async function processIntent(intent, context) {
  const handlers = {
    [INTENTS.RECORD_FOOD]: handleRecordFood,
    [INTENTS.VIEW_REPORT]: handleViewReport,
    [INTENTS.GET_RECOMMEND]: handleGetRecommend,
    [INTENTS.QUERY_NUTRITION]: handleQueryNutrition,
    [INTENTS.CHITCHAT]: handleChitchat,
    [INTENTS.UNKNOWN]: handleUnknown
  }
  
  const handler = handlers[intent.intent] || handlers[INTENTS.UNKNOWN]
  return await handler(context)
}
```

```javascript
// utils/imageProcessor.js

export async function preprocessImage(imagePath) {
  const info = await wx.getImageInfo({ src: imagePath })
  
  let processedPath = imagePath
  
  if (info.width > 800 || info.height > 800) {
    processedPath = await compressImage(imagePath, 800)
  }
  
  const base64 = await imageToBase64(processedPath)
  
  return {
    path: processedPath,
    base64: base64,
    width: info.width,
    height: info.height
  }
}

async function compressImage(path, maxWidth) {
  return new Promise((resolve, reject) => {
    const canvas = wx.createOffscreenCanvas({
      type: '2d',
      width: maxWidth,
      height: maxWidth
    })
    
    const ctx = canvas.getContext('2d')
    const img = canvas.createImage()
    
    img.onload = () => {
      const scale = maxWidth / Math.max(img.width, img.height)
      const width = img.width * scale
      const height = img.height * scale
      
      canvas.width = width
      canvas.height = height
      ctx.drawImage(img, 0, 0, width, height)
      
      wx.canvasToTempFilePath({
        canvas,
        fileType: 'jpg',
        quality: 0.8,
        success: res => resolve(res.tempFilePath),
        fail: reject
      })
    }
    
    img.onerror = reject
    img.src = path
  })
}
```

```javascript
export function calculateBMR(profile) {
  const { weight, height, age, gender } = profile
  
  if (gender === 1) {
    return 10 * weight + 6.25 * height - 5 * age + 5
  } else {
    return 10 * weight + 6.25 * height - 5 * age - 161

export function calculateTDEE(bmr, activityLevel) {
  const multipliers = {
    1: 1.2,
    2: 1.375,
    3: 1.55,
    4: 1.725,
    5: 2.0
  }
```

#### 4.3.2 营养缺口分析

```javascript
export function analyzeNutrientGaps(actual, targets) {
  const gaps = {}
  const surplus = {}
  const adequate = {}
  
  for (const nutrient in targets) {
    const actualValue = actual[nutrient] || 0
    const targetValue = targets[nutrient]
    const ratio = actualValue / targetValue
    
    const result = {
      actual: Math.round(actualValue * 10) / 10,
      target: targetValue,
      ratio: Math.round(ratio * 100)
    }
    
    if (ratio < 0.8) {
      result.gap = Math.round((targetValue - actualValue) * 10) / 10
      result.severity = ratio < 0.5 ? 'high' : 'medium'
      result.status = 'deficient'
      gaps[nutrient] = result
    } else if (ratio > 1.1) {
      result.surplus = Math.round((actualValue - targetValue) * 10) / 10
      result.severity = ratio > 1.3 ? 'high' : 'medium'
      result.status = 'excess'
      surplus[nutrient] = result
    } else {
      result.status = 'adequate'
      adequate[nutrient] = result
    }
  }
  
  return { gaps, surplus, adequate }
}
```

```javascript
export function calculateDailySummary(records) {
  const summary = {
    calories: 0,
    protein: 0,
    fat: 0,
    carbs: 0,
    fiber: 0,
    vitamin_a: 0,
    vitamin_c: 0,
    calcium: 0,
    iron: 0
  }
  
  const byMeal = {
    breakfast: { ...summary },
    lunch: { ...summary },
    dinner: { ...summary },
    snack: { ...summary }
  }
  
  records.forEach(record => {
    const mealType = record.meal_type
    
    record.foods.forEach(food => {
      const multiplier = food.portion / 100
      
      Object.keys(summary).forEach(nutrient => {
        const value = (food.nutrients[nutrient] || 0) * multiplier
        summary[nutrient] += value
        byMeal[mealType][nutrient] += value
      })
    })
  })
  
  Object.keys(summary).forEach(key => {
    summary[key] = Math.round(summary[key] * 10) / 10
  })
  
  return { total: summary, byMeal }
}
```

```javascript
// utils/recommender.js

export async function generateFoodRecommendations(gaps, preferences, database) {
  const recommendations = []
  
  for (const [nutrient, gapInfo] of Object.entries(gaps)) {
    const foods = await findFoodsRichIn(nutrient, database)
    
    const filteredFoods = foods.filter(food => {
      if (preferences.allergies?.includes(food.name)) return false
      if (preferences.dislikes?.includes(food.name)) return false
      return true
    })
    
    const topFoods = filteredFoods
      .sort((a, b) => b.nutrients[nutrient] - a.nutrients[nutrient])
      .slice(0, 3)
    
    topFoods.forEach(food => {
      const suggestedAmount = calculateSuggestedAmount(
        gapInfo.gap,
        food.nutrients[nutrient],
        food.standard_portion
      )
      
      recommendations.push({
        nutrient,
        food: food.name,
        category: food.category,
        suggestedAmount,
        nutrientValue: food.nutrients[nutrient],
        contribution: Math.min(
          (food.nutrients[nutrient] * suggestedAmount / 100) / gapInfo.gap,
          1
        ),
        reason: generateRecommendationReason(nutrient, gapInfo.gap),
        priority: gapInfo.severity === 'high' ? 1 : 2
      })
    })
  }
  
  return recommendations
    .sort((a, b) => a.priority - b.priority)
    .slice(0, 10)
}

function calculateSuggestedAmount(gap, nutrientPer100g, standardPortion) {
  const amountNeeded = (gap / nutrientPer100g) * 100
  return Math.min(Math.ceil(amountNeeded / standardPortion) * standardPortion, 500)
}

function generateRecommendationReason(nutrient, gap) {
  const nutrientNames = {
    vitamin_c: '维生素C',
    calcium: '钙',
    iron: '铁',
    protein: '蛋白质',
    fiber: '膳食纤维'
  }
  
  return `补充${nutrientNames[nutrient] || nutrient} ${Math.round(gap)}mg`
}
```
