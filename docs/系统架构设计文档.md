## 一、架构概述

### 1.1 设计原则

| 原则 | 描述 |
|-----|-----|
| 模块化 | 各功能模块独立开发、独立部署、独立扩展 |
| 可扩展性 | 支持功能扩展和用户规模增长 |
| 高可用性 | 核心功能具备容错能力，保证服务稳定 |
| 安全性 | 数据传输加密，用户隐私保护 |
| 性能优先 | 优化关键路径，保证用户体验 |

### 1.2 架构风格

本项目采用**分层架构 + 微服务思想**的混合架构风格：

- **前端层**：微信小程序客户端
- **网关层**：云函数网关
- **业务层**：云函数服务群
- **数据层**：云数据库 + 云存储
- **外部服务层**：AI识别服务、大模型服务

```javascript
// cloudfunctions/user/index.js

const cloud = require('wx-server-sdk')
cloud.init({ env: cloud.DYNAMIC_CURRENT_ENV })
const db = cloud.database()
const usersCollection = db.collection('users')

exports.main = async (event, context) => {
  const { action, data } = event
  const wxContext = cloud.getWXContext()
  
  switch (action) {
    case 'login':
      return await handleLogin(wxContext.OPENID)
    case 'getProfile':
      return await getProfile(wxContext.OPENID)
    case 'updateProfile':
      return await updateProfile(wxContext.OPENID, data)
    case 'setGoal':
      return await setGoal(wxContext.OPENID, data)
    default:
      return { success: false, error: 'Unknown action' }
  }
}

async function handleLogin(openid) {
  const user = await usersCollection.where({ openid }).get()
  
  if (user.data.length === 0) {
    const newUser = {
      openid,
      nickname: '',
      avatar: '',
      gender: 0,
      age: null,
      height: null,
      weight: null,
      activity_level: 2,
      goal: 'maintain',
      preferences: {
        taste: [],
        dislikes: [],
        allergies: []
      },
      created_at: new Date(),
      updated_at: new Date()
    }
    await usersCollection.add({ data: newUser })
    return { success: true, isNewUser: true, user: newUser }
  }
  
  return { success: true, isNewUser: false, user: user.data[0] }
}
```

**识别服务**

```javascript
// cloudfunctions/recognize/index.js

const cloud = require('wx-server-sdk')
cloud.init({ env: cloud.DYNAMIC_CURRENT_ENV })
const axios = require('axios')

const BAIDU_API_URL = 'https://aip.baidubce.com/rest/2.0/image-classify/v2/food'

exports.main = async (event, context) => {
  const { action, data } = event
  
  switch (action) {
    case 'recognizeFood':
      return await recognizeFood(data.image)
    case 'getNutrient':
      return await getNutrient(data.foodName)
    default:
      return { success: false, error: 'Unknown action' }
  }
}

async function recognizeFood(imageBase64) {
  try {
    const accessToken = await getBaiduAccessToken()
    
    const response = await axios.post(
      `${BAIDU_API_URL}?access_token=${accessToken}`,
      `image=${encodeURIComponent(imageBase64)}`,
      { headers: { 'Content-Type': 'application/x-www-form-urlencoded' } }
    )
    
    const results = response.data.result || []
    
    const enrichedResults = await Promise.all(
      results.map(async (item) => {
        const nutrient = await getNutrient(item.name)
        return {
          name: item.name,
          confidence: item.probability,
          calorie: nutrient?.calorie || 0,
          nutrients: nutrient || {}
        }
      })
    )
    
    return { success: true, foods: enrichedResults }
  } catch (error) {
    console.error('Recognition error:', error)
    return { success: false, error: error.message }
  }
}
```

**营养服务**

```javascript
// cloudfunctions/nutrition/index.js

const cloud = require('wx-server-sdk')
cloud.init({ env: cloud.DYNAMIC_CURRENT_ENV })
const db = cloud.database()

exports.main = async (event, context) => {
  const { action, data } = event
  
  switch (action) {
    case 'calculate':
      return calculateNutrients(data.foods)
    case 'summarize':
      return await summarizeDaily(data.userId, data.date)
    case 'analyzeGap':
      return await analyzeNutrientGap(data.userId, data.date)
    default:
      return { success: false, error: 'Unknown action' }
  }
}

function calculateNutrients(foods) {
  const total = {
    calories: 0,
    protein: 0,
    fat: 0,
    carbs: 0,
    fiber: 0,
    vitamin_a: 0,
    vitamin_c: 0,
    calcium: 0,
    iron: 0
  }
  
  foods.forEach(food => {
    const multiplier = food.portion / 100
    Object.keys(total).forEach(key => {
      if (food.nutrients && food.nutrients[key]) {
        total[key] += food.nutrients[key] * multiplier
      }
    })
  })
  
  return { success: true, total }
}

async function summarizeDaily(userId, date) {
  const records = await db.collection('food_records')
    .where({
      user_id: userId,
      date: date
    })
    .get()
  
  const foods = records.data.flatMap(r => r.foods)
  const total = calculateNutrients(foods)
  
  await db.collection('daily_summary').add({
    data: {
      user_id: userId,
      date: date,
      total_calories: total.total.calories,
      nutrients: total.total,
      created_at: new Date()
    }
  })
  
  return total
}

async function analyzeNutrientGap(userId, date) {
  const user = await db.collection('users').doc(userId).get()
  const summary = await db.collection('daily_summary')
    .where({ user_id: userId, date: date })
    .get()
  
  if (summary.data.length === 0) {
    return { success: false, error: 'No summary found' }
  }
  
  const targets = calculateTargets(user.data)
  const actual = summary.data[0].nutrients
  const gaps = {}
  
  Object.keys(targets).forEach(nutrient => {
    const actualValue = actual[nutrient] || 0
    const targetValue = targets[nutrient]
    const ratio = actualValue / targetValue
    
    if (ratio < 0.8) {
      gaps[nutrient] = {
        actual: actualValue,
        target: targetValue,
        gap: targetValue - actualValue,
        status: 'deficient'
      }
    } else if (ratio > 1.1) {
      gaps[nutrient] = {
        actual: actualValue,
        target: targetValue,
        surplus: actualValue - targetValue,
        status: 'excess'
      }
    }
  })
  
  return { success: true, gaps, targets, actual }
}

function calculateTargets(user) {
  const { weight, height, age, gender, activity_level, goal } = user
  
  let bmr
  if (gender === 1) {
    bmr = 10 * weight + 6.25 * height - 5 * age + 5
  } else {
    bmr = 10 * weight + 6.25 * height - 5 * age - 161
  }
  
  const activityMultipliers = [1.2, 1.375, 1.55, 1.725, 2.0]
  const tdee = bmr * activityMultipliers[activity_level - 1]
  
  let targetCalories
  switch (goal) {
    case 'lose_weight': targetCalories = tdee * 0.8; break
    case 'gain': targetCalories = tdee * 1.1; break
    default: targetCalories = tdee
  }
  
  return {
    calories: Math.round(targetCalories),
    protein: Math.round(targetCalories * 0.2 / 4),
```

```javascript
// utils/aiService.js

class AIService {
  constructor() {
    this.baiduToken = null
    this.tencentToken = null
    this.maxRetries = 3
    this.retryDelay = 1000
  }

  async recognizeFood(imageBase64) {
    const services = [
      () => this.baiduRecognize(imageBase64),
      () => this.tencentRecognize(imageBase64)
    ]
    
    for (const service of services) {
      for (let i = 0; i < this.maxRetries; i++) {
        try {
          const result = await service()
          if (result.success) {
            return result
          }
        } catch (error) {
          console.error(`Service failed (attempt ${i + 1}):`, error)
          if (i < this.maxRetries - 1) {
            await this.delay(this.retryDelay * (i + 1))
          }
        }
      }
    }
    
    return { success: false, error: 'All services failed' }
  }

  async baiduRecognize(imageBase64) {
    const token = await this.getBaiduToken()
    const response = await axios.post(
      `https://aip.baidubce.com/rest/2.0/image-classify/v2/food?access_token=${token}`,
      `image=${encodeURIComponent(imageBase64)}`,
      { headers: { 'Content-Type': 'application/x-www-form-urlencoded' } }
    )
    
    return {
      success: true,
      foods: response.data.result?.map(item => ({
        name: item.name,
        confidence: item.probability,
        calories: item.calorie
      })) || []
    }
  }

  async chatWithAI(message, context) {
    const response = await axios.post(
      'https://aip.baidubce.com/rpc/2.0/ai_custom/v1/wenxinworkshop/chat/completions',
      {
        messages: [
          { role: 'system', content: this.getSystemPrompt() },
          ...context,
          { role: 'user', content: message }
        ]
      },
      {
        headers: {
          'Content-Type': 'application/json',
          'Authorization': `Bearer ${await this.getBaiduToken()}`
        }
      }
    )
    
    return {
      success: true,
      reply: response.data.result
    }
  }

  getSystemPrompt() {
    return `你是一位专业的AI营养师，帮助用户管理饮食健康。
    
你的职责：
1. 识别用户上传的食物图片
2. 分析食物的营养成分
3. 给出个性化的饮食建议
4. 回答用户关于营养和健康的问题

回复要求：
- 友好、专业、简洁
- 提供具体的数值和建议
- 考虑用户的个人情况和目标`
  }

  delay(ms) {
    return new Promise(resolve => setTimeout(resolve, ms))
  }
}

export default new AIService()
```

## 六、性能优化策略

### 6.1 前端优化

| 优化项 | 策略 | 预期效果 |
|-------|-----|---------|

| 懒加载 | 图片/列表懒加载 | 提升页面响应 |
| 防抖节流 | 输入/滚动事件优化 | 减少计算频率 |

### 6.2 后端优化

| 优化项 | 策略 | 预期效果 |
|-------|-----|---------|
| 数据库索引 | 关键字段建立索引 | 查询速度提升80% |
| 结果缓存 | AI识别结果缓存 | 减少API调用 |
| 批量操作 | 批量写入数据库 | 减少IO次数 |
| 异步处理 | 非关键路径异步 | 响应时间减少 |
